## 二、到家服务完整链路测试脚本

**文件路径**: `/root/community-backend/doc/cw测试命令/scripts/test-home-service-full-chain.sh`

```bash
#!/bin/bash
#######################################################################
# 到家服务完整链路测试脚本
# 测试链路: 用户登录 → 绑定小区 → 创建订单 → 支付 → 技工接单 → 
#          建立对话 → 上门打卡 → 完成服务 → 用户确认 → 检查余额
# 执行方式: bash test-home-service-full-chain.sh
#######################################################################

set -e

# 配置
BASE_URL="${BASE_URL:-http://127.0.0.1:3099/api/v1}"
DB_NAME="${DB_NAME:-community_db}"
DB_USER="${DB_USER:-root}"

# 颜色输出
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

log_info() { echo -e "${GREEN}[INFO]${NC} $1"; }
log_warn() { echo -e "${YELLOW}[WARN]${NC} $1"; }
log_error() { echo -e "${RED}[ERROR]${NC} $1"; }
log_step() { echo ""; echo "========================================"; echo "$1"; echo "========================================"; }

# 检查服务是否启动
check_service() {
    log_step "检查服务状态"
    if curl -s "$BASE_URL/../health" > /dev/null 2>&1; then
        log_info "服务已启动"
    else
        log_warn "服务可能未启动，尝试继续..."
    fi
}

# 数据库检查函数
db_check() {
    local sql="$1"
    local desc="$2"
    log_info "数据库检查: $desc"
    mysql -u $DB_USER -p $DB_NAME -e "$sql" 2>/dev/null || log_warn "数据库查询失败，请手动检查"
}

# 主测试流程
main() {
    TIMESTAMP=$(date +%s)
    
    log_step "到家服务完整链路测试开始"
    check_service

    # ==================== 步骤1: 用户登录并绑定小区 ====================
    log_step "步骤1: 用户登录并绑定小区"
    
    log_info "1.1 用户微信授权登录"
    USER_LOGIN_RESPONSE=$(curl -s -X POST "$BASE_URL/auth/wechat/login" \
        -H "Content-Type: application/json" \
        -d "{
            \"code\": \"test_code_user_${TIMESTAMP}\",
            \"userInfo\": {
                \"nickName\": \"测试用户_${TIMESTAMP}\",
                \"avatarUrl\": \"https://example.com/avatar.jpg\"
            }
        }")
    
    USER_TOKEN=$(echo $USER_LOGIN_RESPONSE | grep -o '"token":"[^"]*"' | sed 's/"token":"//;s/"$//')
    USER_ID=$(echo $USER_LOGIN_RESPONSE | grep -o '"id":[0-9]*' | head -1 | sed 's/"id"://')
    
    if [ -z "$USER_TOKEN" ]; then
        log_error "用户登录失败"
        echo "响应: $USER_LOGIN_RESPONSE"
        exit 1
    fi
    log_info "用户登录成功, ID: $USER_ID"
    
    log_info "1.2 绑定小区"
    BIND_RESPONSE=$(curl -s -X POST "$BASE_URL/user/bind-community" \
        -H "Content-Type: application/json" \
        -H "Authorization: Bearer $USER_TOKEN" \
        -d '{"community_id": 1}')
    log_info "绑定小区完成"
    
    # 检查点1
    db_check "SELECT id, nickname, community_id FROM users WHERE id = $USER_ID;" "用户数据入库且绑定小区"

    # ==================== 步骤2: 技工登录 ====================
    log_step "步骤2: 技工登录"
    
    log_info "2.1 技工微信授权登录"
    WORKER_LOGIN_RESPONSE=$(curl -s -X POST "$BASE_URL/auth/wechat/login" \
        -H "Content-Type: application/json" \
        -d "{
            \"code\": \"test_code_worker_${TIMESTAMP}\",
            \"userInfo\": {
                \"nickName\": \"测试技工_${TIMESTAMP}\",
                \"avatarUrl\": \"https://example.com/worker.jpg\"
            }
        }")
    
    WORKER_TOKEN=$(echo $WORKER_LOGIN_RESPONSE | grep -o '"token":"[^"]*"' | sed 's/"token":"//;s/"$//')
    WORKER_ID=$(echo $WORKER_LOGIN_RESPONSE | grep -o '"id":[0-9]*' | head -1 | sed 's/"id"://')
    
    if [ -z "$WORKER_TOKEN" ]; then
        log_error "技工登录失败"
        exit 1
    fi
    log_info "技工登录成功, ID: $WORKER_ID"
    
    log_info "2.2 技工绑定小区"
    curl -s -X POST "$BASE_URL/user/bind-community" \
        -H "Authorization: Bearer $WORKER_TOKEN" \
        -d '{"community_id": 1}' > /dev/null
    log_info "技工绑定小区完成"

    # ==================== 步骤3: 创建订单 ====================
    log_step "步骤3: 创建订单（直约技工）"
    
    log_info "3.1 创建服务订单"
    CREATE_ORDER_RESPONSE=$(curl -s -X POST "$BASE_URL/service-orders" \
        -H "Content-Type: application/json" \
        -H "Authorization: Bearer $USER_TOKEN" \
        -d "{
            \"service_id\": 1,
            \"worker_id\": $WORKER_ID,
            \"community_id\": 1,
            \"address_snapshot\": {
                \"contact\": \"张三\",
                \"phone\": \"13800138000\",
                \"label\": \"上海合川路地铁站\",
                \"detail\": \"1号楼101室\",
                \"latitude\": 31.2304,
                \"longitude\": 121.4737
            },
            \"appointment_time\": \"$(date -d '+1 day' +%Y-%m-%d) 10:00:00\",
            \"remark\": \"请准时上门服务\"
        }")
    
    ORDER_ID=$(echo $CREATE_ORDER_RESPONSE | grep -o '"order_id":[0-9]*' | sed 's/"order_id"://')
    ORDER_NO=$(echo $CREATE_ORDER_RESPONSE | grep -o '"order_no":"[^"]*"' | sed 's/"order_no":"//;s/"$//')
    
    if [ -z "$ORDER_ID" ]; then
        log_error "创建订单失败"
        echo "响应: $CREATE_ORDER_RESPONSE"
        exit 1
    fi
    log_info "订单创建成功, ID: $ORDER_ID, 订单号: $ORDER_NO"
    
    # 检查点2: 订单数据入库
    db_check "SELECT id, order_no, user_id, community_id, assigned_worker_id, status, pay_status, amount FROM service_orders WHERE id = $ORDER_ID;" "订单数据入库"
    
    # 检查点3: 订单绑定到正确的小区
    db_check "SELECT so.id, so.community_id, c.name FROM service_orders so LEFT JOIN communities c ON so.community_id = c.id WHERE so.id = $ORDER_ID;" "订单绑定小区"

    # ==================== 步骤4: 支付订单 ====================
    log_step "步骤4: 支付订单"
    
    log_info "4.1 模拟支付订单"
    PAY_RESPONSE=$(curl -s -X POST "$BASE_URL/service-orders/$ORDER_ID/pay" \
        -H "Authorization: Bearer $USER_TOKEN")
    
    PAY_STATUS=$(echo $PAY_RESPONSE | grep -o '"pay_status":"[^"]*"' | sed 's/"pay_status":"//;s/"$//')
    log_info "支付状态: $PAY_STATUS"
    
    # 检查点4: 支付状态
    db_check "SELECT id, order_no, status, pay_status, amount FROM service_orders WHERE id = $ORDER_ID;" "支付状态验证"

    # ==================== 步骤5: 技工接单 ====================
    log_step "步骤5: 技工接单"
    
    log_info "5.1 技工接单"
    ACCEPT_RESPONSE=$(curl -s -X POST "$BASE_URL/worker/service-orders/$ORDER_ID/accept" \
        -H "Authorization: Bearer $WORKER_TOKEN")
    
    ORDER_STATUS=$(echo $ACCEPT_RESPONSE | grep -o '"status":"[^"]*"' | sed 's/"status":"//;s/"$//')
    log_info "订单状态: $ORDER_STATUS"
    
    # 检查点5: 订单状态变更为服务中
    db_check "SELECT id, order_no, status, assigned_worker_id FROM service_orders WHERE id = $ORDER_ID;" "技工接单状态"

    # ==================== 步骤6: 建立对话 ====================
    log_step "步骤6: 建立对话（用户与技工）"
    
    log_info "6.1 用户发送消息给技工"
    SEND_MSG_RESPONSE=$(curl -s -X POST "$BASE_URL/messages/send" \
        -H "Content-Type: application/json" \
        -H "Authorization: Bearer $USER_TOKEN" \
        -d "{
            \"peerId\": $WORKER_ID,
            \"content\": \"您好，请问什么时候能上门服务？\",
            \"msgType\": \"text\"
        }")
    
    CONVERSATION_ID=$(echo $SEND_MSG_RESPONSE | grep -o '"conversation_id":[0-9]*' | sed 's/"conversation_id"://')
    log_info "会话ID: $CONVERSATION_ID"
    
    # 检查点6: 对话已建立
    if [ -n "$CONVERSATION_ID" ]; then
        db_check "SELECT id, type, last_message_preview FROM conversations WHERE id = $CONVERSATION_ID;" "会话创建"
        db_check "SELECT user_id, conversation_id, peer_id FROM user_conversations WHERE conversation_id = $CONVERSATION_ID;" "用户会话映射"
        db_check "SELECT id, conversation_id, sender_id, content FROM messages WHERE conversation_id = $CONVERSATION_ID;" "消息记录"
    else
        log_warn "会话ID未获取，跳过对话检查"
    fi

    # ==================== 步骤7: 查看地址信息 ====================
    log_step "步骤7: 查看订单地址信息"
    
    log_info "7.1 技工查看订单详情"
    ORDER_DETAIL=$(curl -s -X GET "$BASE_URL/worker/service-orders/$ORDER_ID" \
        -H "Authorization: Bearer $WORKER_TOKEN")
    
    # 检查点7: 地址信息可查看
    db_check "SELECT id, order_no, address_snapshot, contact_name, contact_phone, appointment_time FROM service_orders WHERE id = $ORDER_ID;" "地址信息"

    # ==================== 步骤8: 上门打卡拍照 ====================
    log_step "步骤8: 上门打卡拍照"
    
    log_info "8.1 技工上门打卡"
    CHECKIN_RESPONSE=$(curl -s -X POST "$BASE_URL/worker/service-orders/$ORDER_ID/check-in" \
        -H "Content-Type: application/json" \
        -H "Authorization: Bearer $WORKER_TOKEN" \
        -d '{
            "latitude": 31.2304,
            "longitude": 121.4737,
            "address": "上海市闵行区合川路地铁站1号楼101室"
        }')
    log_info "打卡完成"
    
    log_info "8.2 技工上传服务前证据"
    EVIDENCE_RESPONSE=$(curl -s -X POST "$BASE_URL/worker/service-orders/$ORDER_ID/evidence" \
        -H "Content-Type: application/json" \
        -H "Authorization: Bearer $WORKER_TOKEN" \
        -d '{
            "type": "before",
            "images": ["/uploads/evidence/before_1.jpg", "/uploads/evidence/before_2.jpg"]
        }')
    log_info "证据上传完成"
    
    # 检查点8: 打卡数据入库
    db_check "SELECT id, order_id, check_in_time, latitude, longitude, address FROM service_order_check_ins WHERE order_id = $ORDER_ID;" "打卡记录"
    db_check "SELECT id, order_id, type, images FROM service_order_evidence WHERE order_id = $ORDER_ID;" "证据记录"

    # ==================== 步骤9: 完成服务打卡拍照 ====================
    log_step "步骤9: 完成服务打卡拍照"
    
    log_info "9.1 技工完成服务"
    COMPLETE_RESPONSE=$(curl -s -X POST "$BASE_URL/worker/service-orders/$ORDER_ID/complete" \
        -H "Content-Type: application/json" \
        -H "Authorization: Bearer $WORKER_TOKEN" \
        -d '{
            "images": ["/uploads/evidence/after_1.jpg", "/uploads/evidence/after_2.jpg"],
            "note": "服务已完成，请确认"
        }')
    
    ORDER_STATUS=$(echo $COMPLETE_RESPONSE | grep -o '"status":"[^"]*"' | sed 's/"status":"//;s/"$//')
    log_info "订单状态: $ORDER_STATUS"
    
    # 检查点9: 订单状态变更为待用户确认
    db_check "SELECT id, order_no, status FROM service_orders WHERE id = $ORDER_ID;" "服务完成状态"

    # ==================== 步骤10: 用户确认完成 ====================
    log_step "步骤10: 用户确认完成"
    
    log_info "10.1 用户确认完成"
    CONFIRM_RESPONSE=$(curl -s -X POST "$BASE_URL/service-orders/$ORDER_ID/confirm-complete" \
        -H "Authorization: Bearer $USER_TOKEN")
    
    ORDER_STATUS=$(echo $CONFIRM_RESPONSE | grep -o '"status":"[^"]*"' | sed 's/"status":"//;s/"$//')
    log_info "订单状态: $ORDER_STATUS"
    
    # 检查点10: 订单状态变更为已完成
    db_check "SELECT id, order_no, status, pay_status, amount FROM service_orders WHERE id = $ORDER_ID;" "订单完成状态"

    # ==================== 步骤11: 检查技工余额 ====================
    log_step "步骤11: 检查技工余额"
    
    log_info "11.1 查看技工余额"
    BALANCE_RESPONSE=$(curl -s -X GET "$BASE_URL/user/profile" \
        -H "Authorization: Bearer $WORKER_TOKEN")
    
    # 检查点11: 技工余额
    db_check "SELECT id, nickname, balance FROM users WHERE id = $WORKER_ID;" "技工余额"

    # ==================== 步骤12: 后台查看数据 ====================
    log_step "步骤12: 后台查看订单数据"
    
    log_info "12.1 管理员登录"
    ADMIN_LOGIN_RESPONSE=$(curl -s -X POST "$BASE_URL/auth/admin/login" \
        -H "Content-Type: application/json" \
        -d '{"username": "admin", "password": "admin123"}')
    
    ADMIN_TOKEN=$(echo $ADMIN_LOGIN_RESPONSE | grep -o '"token":"[^"]*"' | sed 's/"token":"//;s/"$//')
    
    if [ -n "$ADMIN_TOKEN" ]; then
        log_info "12.2 管理员查看订单详情"
        curl -s -X GET "$BASE_URL/admin/service-orders/$ORDER_ID" \
            -H "Authorization: Bearer $ADMIN_TOKEN" | head -c 500
        echo ""
        
        log_info "12.3 管理员查看打卡记录"
        curl -s -X GET "$BASE_URL/admin/service-orders/$ORDER_ID/check-ins" \
            -H "Authorization: Bearer $ADMIN_TOKEN" | head -c 500
        echo ""
    else
        log_warn "管理员登录失败，跳过后台检查"
    fi
    
    # 检查点12: 后台可查看完整数据
    db_check "SELECT * FROM service_orders WHERE id = $ORDER_ID\\G" "订单完整信息"

    # ==================== 测试完成 ====================
    log_step "测试完成"
    log_info "到家服务完整链路测试完成"
    log_info "订单ID: $ORDER_ID"
    log_info "订单号: $ORDER_NO"
    log_info "用户ID: $USER_ID"
    log_info "技工ID: $WORKER_ID"
    
    # 输出测试摘要
    echo ""
    echo "==================== 测试摘要 ===================="
    echo "测试时间: $(date)"
    echo "订单ID: $ORDER_ID"
    echo "订单号: $ORDER_NO"
    echo "用户ID: $USER_ID"
    echo "技工ID: $WORKER_ID"
    echo "会话ID: ${CONVERSATION_ID:-未创建}"
    echo "=================================================="
}

# 执行测试
main "$@"
```

## 三、本地集市完整链路测试脚本

**文件路径**: `/root/community-backend/doc/cw测试命令/scripts/test-market-full-chain.sh`

```bash
#!/bin/bash
#######################################################################
# 本地集市完整链路测试脚本
# 测试链路: 商家入驻 → 创建商品(SKU) → 用户下单 → 支付 → 
#          商家接单发货 → 用户确认收货 → 检查余额
# 执行方式: bash test-market-full-chain.sh
#######################################################################

set -e

BASE_URL="${BASE_URL:-http://127.0.0.1:3099/api/v1}"
DB_NAME="${DB_NAME:-community_db}"
DB_USER="${DB_USER:-root}"

RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

log_info() { echo -e "${GREEN}[INFO]${NC} $1"; }
log_warn() { echo -e "${YELLOW}[WARN]${NC} $1"; }
log_error() { echo -e "${RED}[ERROR]${NC} $1"; }
log_step() { echo ""; echo "========================================"; echo "$1"; echo "========================================"; }

db_check() {
    local sql="$1"
    local desc="$2"
    log_info "数据库检查: $desc"
    mysql -u $DB_USER -p $DB_NAME -e "$sql" 2>/dev/null || log_warn "数据库查询失败"
}

main() {
    TIMESTAMP=$(date +%s)
    
    log_step "本地集市完整链路测试开始"

    # ==================== 步骤1: 商家入驻 ====================
    log_step "步骤1: 商家入驻"
    
    log_info "1.1 商家用户登录"
    MERCHANT_LOGIN_RESPONSE=$(curl -s -X POST "$BASE_URL/auth/wechat/login" \
        -H "Content-Type: application/json" \
        -d "{\"code\": \"test_merchant_${TIMESTAMP}\", \"userInfo\": {\"nickName\": \"测试商家_${TIMESTAMP}\"}}")
    
    MERCHANT_TOKEN=$(echo $MERCHANT_LOGIN_RESPONSE | grep -o '"token":"[^"]*"' | sed 's/"token":"//;s/"$//')
    MERCHANT_ID=$(echo $MERCHANT_LOGIN_RESPONSE | grep -o '"id":[0-9]*' | head -1 | sed 's/"id"://')
    log_info "商家ID: $MERCHANT_ID"
    
    log_info "1.2 提交入驻申请"
    MERCHANT_APP_RESPONSE=$(curl -s -X POST "$BASE_URL/market/merchant/apply" \
        -H "Content-Type: application/json" \
        -H "Authorization: Bearer $MERCHANT_TOKEN" \
        -d "{
            \"shop_name\": \"测试店铺_${TIMESTAMP}\",
            \"contact_name\": \"王经理\",
            \"phone\": \"13700137000\",
            \"category\": \"食品生鲜\",
            \"address\": \"上海市闵行区合川路123号\",
            \"entity_name\": \"测试食品有限公司\",
            \"credit_code\": \"91310112MA1GXXXXXX\",
            \"legal_person\": \"王五\"
        }")
    
    MERCHANT_APP_ID=$(echo $MERCHANT_APP_RESPONSE | grep -o '"application_id":[0-9]*' | sed 's/"application_id"://')
    log_info "商家申请ID: $MERCHANT_APP_ID"
    
    log_info "1.3 管理员审核"
    ADMIN_LOGIN_RESPONSE=$(curl -s -X POST "$BASE_URL/auth/admin/login" \
        -H "Content-Type: application/json" \
        -d '{"username": "admin", "password": "admin123"}')
    
    ADMIN_TOKEN=$(echo $ADMIN_LOGIN_RESPONSE | grep -o '"token":"[^"]*"' | sed 's/"token":"//;s/"$//')
    
    curl -s -X PUT "$BASE_URL/admin/market-merchant-applications/$MERCHANT_APP_ID" \
        -H "Authorization: Bearer $ADMIN_TOKEN" \
        -d '{"status": "approved"}' > /dev/null
    log_info "审核通过"
    
    # 检查点1
    db_check "SELECT id, shop_name, status FROM market_shops ORDER BY id DESC LIMIT 1;" "商家入驻成功"

    # ==================== 步骤2: 创建商品 ====================
    log_step "步骤2: 商家创建商品（含SKU）"
    
    log_info "2.1 创建商品"
    CREATE_GOODS_RESPONSE=$(curl -s -X POST "$BASE_URL/market/merchant/goods" \
        -H "Content-Type: application/json" \
        -H "Authorization: Bearer $MERCHANT_TOKEN" \
        -d '{
            "name": "有机苹果",
            "description": "新鲜有机苹果",
            "main_images": ["https://example.com/apple.jpg"],
            "sku_tree": [{"k_id": 1, "k": "规格", "v": [{"id": 1, "name": "1斤装"}, {"id": 2, "name": "2斤装"}]}],
            "sku_list": [
                {"id": "sku_1", "specs": [{"k_id": 1, "v_id": 1}], "price": 15.00, "stock": 100},
                {"id": "sku_2", "specs": [{"k_id": 1, "v_id": 2}], "price": 25.00, "stock": 50}
            ]
        }')
    
    GOODS_ID=$(echo $CREATE_GOODS_RESPONSE | grep -o '"goods_id":[0-9]*' | sed 's/"goods_id"://')
    log_info "商品ID: $GOODS_ID"
    
    log_info "2.2 上架商品"
    curl -s -X POST "$BASE_URL/market/merchant/goods/$GOODS_ID/shelf" \
        -H "Authorization: Bearer $MERCHANT_TOKEN" > /dev/null
    log_info "商品上架成功"
    
    # 检查点2
    db_check "SELECT id, name, status FROM market_goods WHERE id = $GOODS_ID;" "商品创建"
    db_check "SELECT id, goods_id, specs, price, stock FROM market_good_skus WHERE goods_id = $GOODS_ID;" "SKU创建"

    # ==================== 步骤3: 用户下单 ====================
    log_step "步骤3: 用户下单"
    
    log_info "3.1 用户登录"
    USER_LOGIN_RESPONSE=$(curl -s -X POST "$BASE_URL/auth/wechat/login" \
        -H "Content-Type: application/json" \
        -d "{\"code\": \"test_buyer_${TIMESTAMP}\", \"userInfo\": {\"nickName\": \"买家_${TIMESTAMP}\"}}")
    
    USER_TOKEN=$(echo $USER_LOGIN_RESPONSE | grep -o '"token":"[^"]*"' | sed 's/"token":"//;s/"$//')
    USER_ID=$(echo $USER_LOGIN_RESPONSE | grep -o '"id":[0-9]*' | head -1 | sed 's/"id"://')
    
    log_info "3.2 创建订单"
    CREATE_ORDER_RESPONSE=$(curl -s -X POST "$BASE_URL/market/orders" \
        -H "Content-Type: application/json" \
        -H "Authorization: Bearer $USER_TOKEN" \
        -d "{
            \"shop_id\": 1,
            \"delivery_mode\": \"express\",
            \"address\": {
                \"contact\": \"张三\",
                \"phone\": \"13800138000\",
                \"province\": \"上海市\",
                \"city\": \"上海市\",
                \"district\": \"闵行区\",
                \"detail\": \"合川路123号\"
            },
            \"items\": [{\"goods_id\": $GOODS_ID, \"sku_id\": \"sku_1\", \"quantity\": 2}]
        }")
    
    ORDER_NO=$(echo $CREATE_ORDER_RESPONSE | grep -o '"order_no":"[^"]*"' | sed 's/"order_no":"//;s/"$//')
    log_info "订单号: $ORDER_NO"
    
    # 检查点3
    db_check "SELECT order_no, order_status, pay_status, amount FROM market_orders WHERE order_no = '$ORDER_NO';" "订单创建"
    db_check "SELECT order_no, goods_id, sku_id, quantity, price FROM market_order_items WHERE order_no = '$ORDER_NO';" "订单明细"

    # ==================== 步骤4: 支付订单 ====================
    log_step "步骤4: 支付订单"
    
    log_info "4.1 模拟支付"
    curl -s -X POST "$BASE_URL/market/payments/mock-success" \
        -H "Authorization: Bearer $USER_TOKEN" \
        -d "{\"order_no\": \"$ORDER_NO\"}" > /dev/null
    log_info "支付成功"
    
    # 检查点4
    db_check "SELECT order_no, order_status, pay_status FROM market_orders WHERE order_no = '$ORDER_NO';" "支付状态"
    db_check "SELECT * FROM market_payments WHERE order_no = '$ORDER_NO';" "支付记录"

    # ==================== 步骤5: 商家接单发货 ====================
    log_step "步骤5: 商家接单发货"
    
    log_info "5.1 商家接单"
    curl -s -X POST "$BASE_URL/market/merchant/orders/$ORDER_NO/accept" \
        -H "Authorization: Bearer $MERCHANT_TOKEN" > /dev/null
    
    log_info "5.2 商家发货"
    curl -s -X POST "$BASE_URL/market/merchant/orders/$ORDER_NO/dispatch" \
        -H "Content-Type: application/json" \
        -H "Authorization: Bearer $MERCHANT_TOKEN" \
        -d '{"express_company": "顺丰快递", "express_no": "SF1234567890"}' > /dev/null
    log_info "发货成功"
    
    # 检查点5
    db_check "SELECT order_no, order_status, express_company, express_no FROM market_orders WHERE order_no = '$ORDER_NO';" "发货状态"

    # ==================== 步骤6: 用户确认收货 ====================
    log_step "步骤6: 用户确认收货"
    
    log_info "6.1 确认收货"
    curl -s -X POST "$BASE_URL/market/orders/$ORDER_NO/confirm" \
        -H "Authorization: Bearer $USER_TOKEN" > /dev/null
    log_info "确认收货成功"
    
    # 检查点6
    db_check "SELECT order_no, order_status FROM market_orders WHERE order_no = '$ORDER_NO';" "订单完成"

    # ==================== 步骤7: 检查库存和余额 ====================
    log_step "步骤7: 检查库存扣减和商家余额"

    # 检查点8: 商家余额
    db_check "SELECT id, nickname, balance FROM users WHERE id = $MERCHANT_ID;" "商家余额"

    # ==================== 测试完成 ====================
    log_step "测试完成"
    log_info "本地集市完整链路测试完成"
    
    echo ""
    echo "==================== 测试摘要 ===================="
    echo "测试时间: $(date)"
    echo "商家ID: $MERCHANT_ID"
    echo "商品ID: $GOODS_ID"
    echo "订单号: $ORDER_NO"
    echo "用户ID: $USER_ID"

}
```

## 四、邻里帮帮完整链路测试脚本

**文件路径**: `/root/community-backend/doc/cw测试命令/scripts/test-neighbor-assist-full-chain.sh`

```bash
#!/bin/bash
#######################################################################
# 邻里帮帮完整链路测试脚本
# 测试链路: 用户发布需求 → 支付 → 进入待派单池 → 技工抢单 → 
#          完成服务 → 检查余额
# 执行方式: bash test-neighbor-assist-full-chain.sh
#######################################################################

set -e

BASE_URL="${BASE_URL:-http://127.0.0.1:3099/api/v1}"
DB_NAME="${DB_NAME:-community_db}"
DB_USER="${DB_USER:-root}"

RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

log_info() { echo -e "${GREEN}[INFO]${NC} $1"; }
log_warn() { echo -e "${YELLOW}[WARN]${NC} $1"; }
log_error() { echo -e "${RED}[ERROR]${NC} $1"; }
log_step() { echo ""; echo "========================================"; echo "$1"; echo "========================================"; }

db_check() {
    local sql="$1"
    local desc="$2"
    log_info "数据库检查: $desc"
    mysql -u $DB_USER -p $DB_NAME -e "$sql" 2>/dev/null || log_warn "数据库查询失败"
}

main() {
    TIMESTAMP=$(date +%s)
    
    log_step "邻里帮帮完整链路测试开始"

    # ==================== 步骤1: 用户发布需求 ====================
    log_step "步骤1: 用户发布帮帮需求"
    
    log_info "1.1 用户登录"
    USER_LOGIN_RESPONSE=$(curl -s -X POST "$BASE_URL/auth/wechat/login" \
        -H "Content-Type: application/json" \
        -d "{\"code\": \"test_assist_user_${TIMESTAMP}\", \"userInfo\": {\"nickName\": \"帮帮用户_${TIMESTAMP}\"}}")
    
    USER_TOKEN=$(echo $USER_LOGIN_RESPONSE | grep -o '"token":"[^"]*"' | sed 's/"token":"//;s/"$//')
    USER_ID=$(echo $USER_LOGIN_RESPONSE | grep -o '"id":[0-9]*' | head -1 | sed 's/"id"://')
    
    log_info "1.2 绑定小区"
    curl -s -X POST "$BASE_URL/user/bind-community" \
        -H "Authorization: Bearer $USER_TOKEN" \
        -d '{"community_id": 1}' > /dev/null
    
    log_info "1.3 发布帮帮需求"
    CREATE_RESPONSE=$(curl -s -X POST "$BASE_URL/neighbor-assist/orders" \
        -H "Content-Type: application/json" \
        -H "Authorization: Bearer $USER_TOKEN" \
        -d '{
            "assist_type": "take",
            "community_id": 1,
            "origin_address_snapshot": {
                "label": "上海合川路地铁站",
                "detail": "快递柜A区",
                "contact": "张三",
                "phone": "13800138000"
            },
            "destination_address_snapshot": {
                "label": "上海合川路地铁站",
                "detail": "1号楼101",
                "contact": "李四",
                "phone": "13900139000"
            },
            "amount": 10.5,
            "remark": "请小心轻放"
        }')
    
    ORDER_ID=$(echo $CREATE_RESPONSE | grep -o '"order_id":[0-9]*' | sed 's/"order_id"://')
    log_info "帮帮订单ID: $ORDER_ID"
    
    # 检查点1
    db_check "SELECT id, assist_type, status, pay_status, amount FROM neighbor_assist_orders WHERE id = $ORDER_ID;" "需求创建"

    # ==================== 步骤2: 支付报酬 ====================
    log_step "步骤2: 支付报酬"
    
    curl -s -X POST "$BASE_URL/neighbor-assist/orders/$ORDER_ID/pay" \
        -H "Authorization: Bearer $USER_TOKEN" > /dev/null
    log_info "支付成功"
    
    # 检查点2
    db_check "SELECT id, status, pay_status FROM neighbor_assist_orders WHERE id = $ORDER_ID;" "支付状态"

    # ==================== 步骤3: 进入待派单池 ====================
    log_step "步骤3: 验证订单进入待派单池"
    
    log_info "3.1 技工登录"
    WORKER_LOGIN_RESPONSE=$(curl -s -X POST "$BASE_URL/auth/wechat/login" \
        -H "Content-Type: application/json" \
        -d "{\"code\": \"test_assist_worker_${TIMESTAMP}\", \"userInfo\": {\"nickName\": \"帮帮技工_${TIMESTAMP}\"}}")
    
    WORKER_TOKEN=$(echo $WORKER_LOGIN_RESPONSE | grep -o '"token":"[^"]*"' | sed 's/"token":"//;s/"$//')
    WORKER_ID=$(echo $WORKER_LOGIN_RESPONSE | grep -o '"id":[0-9]*' | head -1 | sed 's/"id"://')
    
    curl -s -X POST "$BASE_URL/user/bind-community" \
        -H "Authorization: Bearer $WORKER_TOKEN" \
        -d '{"community_id": 1}' > /dev/null
    
    log_info "3.2 技工查看待派单池"
    POOL_RESPONSE=$(curl -s -X GET "$BASE_URL/neighbor-assist/orders/pool?limit=50" \
        -H "Authorization: Bearer $WORKER_TOKEN")
    
    # 检查点3
    if echo "$POOL_RESPONSE" | grep -q "\"id\":$ORDER_ID"; then
        log_info "✓ 订单在待派单池中"
    else
        log_warn "✗ 订单不在待派单池中"
    fi

    # ==================== 步骤4: 技工抢单 ====================
    log_step "步骤4: 技工抢单"
    
    GRAB_RESPONSE=$(curl -s -X POST "$BASE_URL/neighbor-assist/orders/$ORDER_ID/grab" \
        -H "Authorization: Bearer $WORKER_TOKEN")
    log_info "抢单完成"
    
    # 检查点4
    db_check "SELECT id, status, assigned_worker_id FROM neighbor_assist_orders WHERE id = $ORDER_ID;" "抢单状态"
    db_check "SELECT * FROM neighbor_assist_grab_records WHERE order_id = $ORDER_ID;" "抢单记录"

    # ==================== 步骤5: 技工完成服务 ====================
    log_step "步骤5: 技工完成服务"
    
    curl -s -X POST "$BASE_URL/neighbor-assist/orders/$ORDER_ID/accept" \
        -H "Authorization: Bearer $WORKER_TOKEN" > /dev/null
    
    curl -s -X POST "$BASE_URL/neighbor-assist/orders/$ORDER_ID/complete" \
        -H "Authorization: Bearer $WORKER_TOKEN" > /dev/null
    log_info "服务完成"
    
    # 检查点5
    db_check "SELECT id, status FROM neighbor_assist_orders WHERE id = $ORDER_ID;" "服务完成"

    # ==================== 步骤6: 检查技工余额 ====================
    log_step "步骤6: 检查技工余额"
    
    db_check "SELECT id, nickname, balance FROM users WHERE id = $WORKER_ID;" "技工余额"

    # ==================== 测试完成 ====================
    log_step "测试完成"
    
    echo ""
    echo "==================== 测试摘要 ===================="
    echo "测试时间: $(date)"
    echo "帮帮订单ID: $ORDER_ID"
    echo "用户ID: $USER_ID"
    echo "技工ID: $WORKER_ID"
    echo "=================================================="
}

main "$@"
```
