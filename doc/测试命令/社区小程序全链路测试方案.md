## 一、测试目标与范围

### 1.1 测试目标

本测试方案旨在验证社区小程序全链路业务功能的正确性，确保：
- **数据一致性**: 前端操作 → 后端处理 → 数据库持久化 → 前端展示 全程数据一致
- **角色权限**: 各角色（用户/商家/技工/服务商/骑手）只能访问对应权限的接口
- **业务闭环**: 每个业务流程从发起至完成形成完整闭环
- **消息通知**: 订单状态变化时，相关角色收到正确的消息通知
- **数据隔离**: 不同小区用户数据严格隔离

### 1.2 测试范围与优先级

| 模块 | 优先级 | 测试重点 | 预计用例数 |
|-----|--------|---------|-----------|

| 市场购物 | P0 | 浏览→购物车→下单→支付→商家接单→发货→收货 | 20 |
| 服务预约 | P0 | 创建订单→服务商接单→派单→技工履约→确认 | 18 |

const http = require('http');

const BASE_URL = 'http://8.136.29.208:3001';
const RESULTS = { pass: 0, fail: 0, warn: 0, err: [] };

function api(method, path, body, token) {
  return new Promise((resolve, reject) => {
    const u = new URL(path, BASE_URL);
    const o = {
      hostname: u.hostname,
      port: u.port,
      path: u.pathname + u.search,
      method,
      headers: {
        'Content-Type': 'application/json',
        'Accept': 'application/json'
      }
    };
    if (token) o.headers['Authorization'] = `Bearer ${token}`;
    
    const req = http.request(o, (res) => {
      let data = '';
      res.on('data', chunk => { data += chunk; });
      res.on('end', () => {
        try {
          resolve({ s: res.statusCode, d: data ? JSON.parse(data) : {}, r: data });
        } catch(e) {
          resolve({ s: res.statusCode, d: {}, r: data });
        }
      });
    });
    req.on('error', reject);
    if (body) req.write(JSON.stringify(body));
    req.end();
  });
}

function report(name, status, msg) {
  const c = status==='PASS'?'32':status==='FAIL'?'31':status==='WARN'?'33':'36';
  console.log(`\x1b[${c}m[${status}] ${name}: ${msg}\x1b[0m`);
  if (status==='PASS') RESULTS.pass++;
  else if (status==='FAIL') { RESULTS.fail++; RESULTS.err.push({t:name, e:msg}); }
  else if (status==='WARN') RESULTS.warn++;
}

async function runNeighborAssistTests() {
  console.log('=== 邻里帮帮全链路测试 ===\n');
  
  // 1. 用户A登录

  report('用户A登录', loginA.s === 200 ? 'PASS' : 'FAIL', loginA.s === 200 ? '登录成功' : loginA.r);
  
  // 2. 发布帮帮订单
  const createOrder = await api('POST', '/neighbor-assist', {

  }, tokenA);
  const orderId = createOrder.d?.id || createOrder.d?.data?.id;
  report('发布帮帮订单', createOrder.s === 200 ? 'PASS' : 'FAIL', `订单ID: ${orderId}`);
  
  // 3. 用户B登录
  const loginB = await api('POST', '/auth/wechat/login', { code: 'test_code_13800138002' });
  const tokenB = loginB.d?.token || loginB.d?.data?.token;
  report('用户B登录', loginB.s === 200 ? 'PASS' : 'FAIL', loginB.s === 200 ? '登录成功' : loginB.r);

  report('查看订单池', Array.isArray(orders) && orders.length > 0 ? 'PASS' : 'FAIL', `订单数: ${orders.length}`);

## 七、问题排查指南

### 7.1 常见问题

| 问题 | 可能原因 | 解决方案 |
|-----|---------|---------|
| 订单发布成功但数据库无记录 | 后端未持久化 | 检查controller是否有db.save() |
| 抢单后发布人无通知 | 消息系统未触发 | 检查订单状态变化时是否调用消息服务 |
| 跨小区能看到订单 | 查询未过滤community_id | 检查SQL查询是否有WHERE community_id=? |
| 订单状态无法流转 | 状态机逻辑错误 | 检查状态转换条件是否满足 |
| 图片上传失败 | 路径配置错误 | 检查uploads目录权限和配置 |
