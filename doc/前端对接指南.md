### 1.2 请求封装

```javascript
// utils/request.js
import ENV from '../config'

const request = (options) => {
  return new Promise((resolve, reject) => {
    const token = wx.getStorageSync('token')
    
    wx.request({
      url: ENV.baseURL + options.url,
      method: options.method || 'GET',
      data: options.data,
      header: {
        'Content-Type': 'application/json',
        'Authorization': token ? `Bearer ${token}` : '',
        ...options.header
      },
      success: (res) => {
        if (res.data.code === 0) {
          resolve(res.data.data)
        } else if (res.data.code === 401) {
          // Token过期，跳转登录
          wx.removeStorageSync('token')
          wx.navigateTo({ url: '/pages/login/login' })
          reject(res.data)
        } else {
          reject(res.data)
        }
      },
      fail: reject
    })
  })
}

export default request
```

### 2.1 微信登录

```javascript
// pages/login/login.js
import request from '../../utils/request'

Page({
  data: {
    canIUseGetUserProfile: false
  },

  onLoad() {
    if (wx.getUserProfile) {
      this.setData({ canIUseGetUserProfile: true })
    }
  },

  // 微信登录
  async wxLogin() {
    try {
      // 1. 获取微信code
      const { code } = await wx.login()
      
      // 2. 获取用户信息
      const { userInfo } = await wx.getUserProfile({
        desc: '用于完善用户资料'
      })
      
      // 3. 调用后端登录接口
      const data = await request({
        url: '/auth/wechat/login',
        method: 'POST',
        data: {
          code: code,
          nickname: userInfo.nickName,
          avatar_url: userInfo.avatarUrl
        }
      })
      
      // 4. 保存token
      wx.setStorageSync('token', data.token)
      wx.setStorageSync('userInfo', data.user)
      
      // 5. 跳转首页
      wx.switchTab({ url: '/pages/index/index' })
      
    } catch (error) {
      console.error('登录失败', error)
      wx.showToast({ title: '登录失败', icon: 'none' })
    }
  },

  // 账号密码登录
  async accountLogin(e) {
    try {
      const { phone, password } = e.detail.value
      
      const data = await request({
        url: '/auth/login',
        method: 'POST',
        data: { phone, password }
      })
      
      wx.setStorageSync('token', data.token)
      wx.setStorageSync('userInfo', data.user)
      
      wx.switchTab({ url: '/pages/index/index' })
      
    } catch (error) {
      wx.showToast({ title: error.message || '登录失败', icon: 'none' })
    }
  }
})
```

### 5.2 购物车页面

```javascript
// pages/market/cart/cart.js
import { getCart, updateCartItem, deleteCartItem } from '../../../api/market'

Page({
  data: {
    cartItems: [],
    totalPrice: 0,
    selectedIds: []
  },

  onShow() {
    this.loadCart()
  },

  async loadCart() {
    try {
      const data = await getCart()
      this.setData({
        cartItems: data.items || [],
        totalPrice: data.total_price || 0
      })
    } catch (error) {
      console.error('获取购物车失败', error)
    }
  },

  async updateQuantity(e) {
    const { id, type } = e.currentTarget.dataset
    const item = this.data.cartItems.find(i => i.id === id)
    
    let quantity = item.quantity
    if (type === 'add') quantity++
    if (type === 'minus') quantity--
    
    if (quantity < 1) {
      await this.deleteItem(id)
      return
    }
    
    try {
      await updateCartItem(id, { quantity })
      this.loadCart()
    } catch (error) {
      wx.showToast({ title: '更新失败', icon: 'none' })
    }
  },

  async deleteItem(id) {
    try {
      await wx.showModal({
        title: '提示',
        content: '确定删除该商品吗？'
      })
      
      await deleteCartItem(id)
      this.loadCart()
      
    } catch (error) {
      // 用户取消
    }
  },

  submitOrder() {
    if (this.data.selectedIds.length === 0) {
      wx.showToast({ title: '请选择商品', icon: 'none' })
      return
    }
    
    wx.navigateTo({
      url: `/pages/market/checkout/checkout?ids=${this.data.selectedIds.join(',')}`
    })
  }
})
```

### 7.2 WebSocket连接

```javascript
// utils/websocket.js
import ENV from '../config'

class WebSocketManager {
  constructor() {
    this.socket = null
    this.isConnected = false
    this.listeners = {}
  }

  connect() {
    const token = wx.getStorageSync('token')
    
    this.socket = wx.connectSocket({
      url: `${ENV.wsURL}/ws?token=${token}`,
      success: () => {
        console.log('WebSocket连接成功')
      }
    })

    wx.onSocketOpen(() => {
      this.isConnected = true
      console.log('WebSocket已打开')
    })

    wx.onSocketMessage((res) => {
      const data = JSON.parse(res.data)
      this.handleMessage(data)
    })

    wx.onSocketClose(() => {
      this.isConnected = false
      console.log('WebSocket已关闭')
      // 自动重连
      setTimeout(() => this.connect(), 3000)
    })
  }

  handleMessage(data) {
    const { type, payload } = data
    
    if (this.listeners[type]) {
      this.listeners[type].forEach(callback => callback(payload))
    }
  }

  on(type, callback) {
    if (!this.listeners[type]) {
      this.listeners[type] = []
    }
    this.listeners[type].push(callback)
  }

  off(type, callback) {
    if (this.listeners[type]) {
      this.listeners[type] = this.listeners[type].filter(cb => cb !== callback)
    }
  }

  send(data) {
    if (this.isConnected) {
      wx.sendSocketMessage({
        data: JSON.stringify(data)
      })
    }
  }
}

export default new WebSocketManager()
```

```javascript
// utils/upload.js
import ENV from '../config'

export const uploadFile = (filePath) => {
  return new Promise((resolve, reject) => {
    const token = wx.getStorageSync('token')
    
    wx.uploadFile({
      url: ENV.baseURL + '/upload',
      filePath: filePath,
      name: 'file',
      header: {
        'Authorization': `Bearer ${token}`
      },
      success: (res) => {
        const data = JSON.parse(res.data)
        if (data.code === 0) {
          resolve(data.url)
        } else {
          reject(data)
        }
      },
      fail: reject
    })
  })
}
```

## 九、错误处理

### 9.1 统一错误处理

```javascript
// utils/errorHandler.js

const ERROR_MESSAGES = {
  '-1': '系统错误，请稍后重试',
  '401': '登录已过期，请重新登录',
  '403': '无权限访问',
  '404': '资源不存在',
  '422': '参数验证失败',
  '500': '服务器错误'
}

export const handleError = (error) => {
  console.error('API Error:', error)
  
  const message = error.message || ERROR_MESSAGES[error.code] || '未知错误'
  
  wx.showToast({
    title: message,
    icon: 'none',
    duration: 2000
  })
  
  // 401错误跳转登录
  if (error.code === 401) {
    setTimeout(() => {
      wx.navigateTo({ url: '/pages/login/login' })
    }, 2000)
  }
  
  return Promise.reject(error)
}
```

### 10.2 数据缓存

```javascript
// 使用小程序缓存
export const setCache = (key, data, expireTime = 3600000) => {
  wx.setStorageSync(key, {
    data,
    expireTime: Date.now() + expireTime
  })
}

export const getCache = (key) => {
  const cache = wx.getStorageSync(key)
  if (!cache) return null
  
  if (Date.now() > cache.expireTime) {
    wx.removeStorageSync(key)
    return null
  }
  
  return cache.data
}
```

### 10.3 请求节流

```javascript
  if (isSubmitting) {
    wx.showToast({ title: '请勿重复提交', icon: 'none' })
    return
  }
  
  isSubmitting = true
  try {
    await fn()
  } finally {

}
```
